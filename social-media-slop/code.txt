    public SlopGenerator(ChatClient.Builder builder) {
        this.chatClient = builder.build();
    }

    /**
     * Generate a tweet based on the provided template.
     * Example: GET /tweets?template=hello
     * @param template the template string
     * @return generated tweet
     */
    @GetMapping("/tweets")
    public String tweets(@RequestParam String originalTweet) {

    	String systemTemplate = """
    	You are a tweet rewrite engine that does not use em-dashes and prefers to end sentences rather than use semi-colons.

    	Goal:
    	Given one ORIGINAL_TWEET, produce FIVE rewritten tweets, each in a different voice:
    	1) PIRATE
    	2) INSPIRATIONAL
    	3) TECH_BRO
    	4) IMPOSTER
    	5) MONK

    	Modernization:
    	Refresh dated concepts to the degree specified by the MODERNIZATION_LEVEL
    	
    	Hard rules:
    	- Output MUST contain exactly five tweets, one per voice, and nothing else.
    	- Add emojis sporadically inside the text each tweet, per the EMOJI_LEVEL (low=1-2 emojis per tweet, med=3-5, high=8-10 emojis per tweet).

    	""";

    	// IMPORTANT: use {var} placeholders (not {{var}}) to match Spring AI PromptTemplate in many setups
    	String userTemplate = """
    	ORIGINAL_TWEET:
    	{postText}

    	TOPIC_HINT:
    	{topicHint}

    	EMOJI_LEVEL:
    	{emojiLevel}

    	MODERNIZATION_LEVEL:
    	{modernizationLevel}
    	""";

    	originalTweet = "How to Realistically Start: ◽HTML ◽CSS ◽JavaScript ◽React ◽Github ◽Git ◽Create a Twitter account  ◽Share Knowledge  ◽Build Network  ◽Get a remote Job";
    	String topicHint = "Lean towards AWS certification.";
    	String emojiLevel = "high";          // low | med | high
    	String modernizationLevel = "high"; // low | med | high

    	PromptTemplate promptTemplate = new PromptTemplate(userTemplate);

    	String userPrompt = promptTemplate.render(Map.of(
    	    "postText", originalTweet,
    	    "topicHint", topicHint,
    	    "emojiLevel", emojiLevel,
    	    "modernizationLevel", modernizationLevel
    	));

    	// Call the model with system + user
    	return chatClient
    	    .prompt()
    	    .system(systemTemplate)
    	    .user(userPrompt)
    	    .call()
    	    .content();
    }
}
